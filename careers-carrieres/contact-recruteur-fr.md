<script>
(function () {
  "use strict";

  var form           = document.getElementById("cr-finder-form");
  var select          = document.getElementById("cr-province-select");
  var resetBtn        = document.getElementById("cr-reset-btn");
  var provincesWrap   = document.getElementById("cr-provinces");
  var allSections     = Array.prototype.slice.call(document.querySelectorAll(".cr-province"));
  var noMatch         = document.getElementById("cr-no-match");
  var liveRegion      = document.getElementById("cr-live-region");

  if (!form || !select || !allSections.length) { return; }

  function announce(msg) {
    liveRegion.textContent = "";
    setTimeout(function () { liveRegion.textContent = msg; }, 50);
  }

  function labelFor(id) {
    var opt = select.querySelector('option[value="' + id + '"]');
    return opt ? opt.textContent : "";
  }

  function hideAll() {
    allSections.forEach(function (section) { section.classList.add("hidden"); });
    if (noMatch) { noMatch.classList.add("hidden"); }
  }

  function showOnly(id) {
    var found = false;
    allSections.forEach(function (section) {
      if (section.id === id) {
        section.classList.remove("hidden");
        found = true;
      } else {
        section.classList.add("hidden");
      }
    });
    noMatch.classList.toggle("hidden", found);
    return found;
  }

  function showAll() {
    allSections.forEach(function (section) { section.classList.remove("hidden"); });
    noMatch.classList.add("hidden");
  }

  // État par défaut : aucun panneau de province/territoire n'est affiché
  // tant que l'utilisateur n'a pas fait de sélection ou demandé de tout afficher.
  hideAll();

  form.addEventListener("submit", function (e) {
    e.preventDefault();
    var id = select.value;
    if (!id) {
      announce("Veuillez d'abord sélectionner une province ou un territoire.");
      select.focus();
      return;
    }
    var found = showOnly(id);
    var target = document.getElementById(id);
    if (found && target) {
      target.scrollIntoView({ behavior: "smooth", block: "start" });
      target.focus();
      announce("Affichage des coordonnées pour " + labelFor(id) + ".");
    } else {
      announce("Aucune coordonnée trouvée pour " + labelFor(id) + ".");
    }
  });

  resetBtn.addEventListener("click", function () {
    showAll();
    select.value = "";
    announce("Affichage des coordonnées de toutes les provinces et de tous les territoires.");
    provincesWrap.scrollIntoView({ behavior: "smooth", block: "start" });
    select.focus();
  });
}());
</script>
